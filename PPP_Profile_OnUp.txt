# 延时 2 秒
delay 2s
# 要调用的 DDNS 脚本名称
:local scrName "CloudFlare_DDNS"
# 此接口使用的域名
:local cfDomain "yd.yourdomain.com"
# 此接口的运营商名称
:local ISP "移动"
# 获取拨号接口名称
:local ifname [/interface get $interface name]
# 获取公网IP
:local currentIP $"local-address"
# 以函数形式调用脚本，不再使用 execute 命令（权限受限）
:local ddnsFunc [:parse [/system script get $scrName source]]
$ddnsFunc cfDomain=$cfDomain ISP=$ISP ifname=$ifname currentIP=$currentIP
